// =====================================================================
// SECTION 1: BASIC PRICE MEASURES
// =====================================================================

// 1.1 Current Prices by Fuel Type

Current PMS Price = 
CALCULATE(
    AVERAGE(fact_fuel_prices[price]),
    FILTER(dim_fuel, dim_fuel[fuel_code] = "PMS"),
    FILTER(
        fact_fuel_prices,
        DATEDIFF(
            fact_fuel_prices[effective_from],
            MAX(dim_date[full_date]),
            DAY
        ) >= 0
        && DATEDIFF(
            fact_fuel_prices[effective_from],
            MAX(dim_date[full_date]),
            DAY
        ) <= 30
    )
)

Current Diesel Price = 
CALCULATE(
    AVERAGE(fact_fuel_prices[price]),
    FILTER(dim_fuel, dim_fuel[fuel_code] = "AGO"),
    FILTER(
        fact_fuel_prices,
        DATEDIFF(
            fact_fuel_prices[effective_from],
            MAX(dim_date[full_date]),
            DAY
        ) >= 0
        && DATEDIFF(
            fact_fuel_prices[effective_from],
            MAX(dim_date[full_date]),
            DAY
        ) <= 30
    )
)

Current Kerosene Price = 
CALCULATE(
    AVERAGE(fact_fuel_prices[price]),
    FILTER(dim_fuel, dim_fuel[fuel_code] = "IK"),
    FILTER(
        fact_fuel_prices,
        DATEDIFF(
            fact_fuel_prices[effective_from],
            MAX(dim_date[full_date]),
            DAY
        ) >= 0
        && DATEDIFF(
            fact_fuel_prices[effective_from],
            MAX(dim_date[full_date]),
            DAY
        ) <= 30
    )
)

Current Average Price = 
CALCULATE(
    AVERAGE(fact_fuel_prices[price]),
    FILTER(
        fact_fuel_prices,
        DATEDIFF(
            fact_fuel_prices[effective_from],
            MAX(dim_date[full_date]),
            DAY
        ) >= 0
        && DATEDIFF(
            fact_fuel_prices[effective_from],
            MAX(dim_date[full_date]),
            DAY
        ) <= 30
    )
)

// 1.2 Historical Prices (Previous Cycle)

Previous PMS Price = 
CALCULATE(
    [Current PMS Price],
    DATEADD(dim_date[full_date], -30, DAY)
)

Previous Diesel Price = 
CALCULATE(
    [Current Diesel Price],
    DATEADD(dim_date[full_date], -30, DAY)
)

Previous Kerosene Price = 
CALCULATE(
    [Current Kerosene Price],
    DATEADD(dim_date[full_date], -30, DAY)
)

Previous Average Price = 
CALCULATE(
    [Current Average Price],
    DATEADD(dim_date[full_date], -30, DAY)
)

// =====================================================================
// SECTION 2: MONTH-OVER-MONTH (MOM) CHANGE MEASURES
// =====================================================================

MoM Change PMS % = 
DIVIDE(
    [Current PMS Price] - [Previous PMS Price],
    [Previous PMS Price],
    0
) * 100

MoM Change Diesel % = 
DIVIDE(
    [Current Diesel Price] - [Previous Diesel Price],
    [Previous Diesel Price],
    0
) * 100

MoM Change Kerosene % = 
DIVIDE(
    [Current Kerosene Price] - [Previous Kerosene Price],
    [Previous Kerosene Price],
    0
) * 100

MoM Change Average % = 
DIVIDE(
    [Current Average Price] - [Previous Average Price],
    [Previous Average Price],
    0
) * 100

// Direction Indicators (for conditional formatting)
MoM Direction = 
IF([MoM Change Average %] > 0, "↑", IF([MoM Change Average %] < 0, "↓", "→"))

// =====================================================================
// SECTION 3: RANKING & PERCENTILE MEASURES
// =====================================================================

Price Percentile Rank = 
PERCENTRANK.INC(
    ALLSELECTED(dim_location[town]),
    [Current Average Price]
) * 100

// Rank towns by price (1 = most expensive)
Price Rank = 
RANKX(
    ALLSELECTED(dim_location[town]),
    [Current Average Price],
    ,
    DESC
)

// Rank towns by price (1 = cheapest)
Price Rank Ascending = 
RANKX(
    ALLSELECTED(dim_location[town]),
    [Current Average Price],
    ,
    ASC
)

// Percentile bucket (Top 25%, Top 50%, Bottom 50%, Bottom 25%)
Price Percentile Bucket = 
IF(
    [Price Percentile Rank] >= 75,
    "Top 25% (Most Expensive)",
    IF(
        [Price Percentile Rank] >= 50,
        "50-75%",
        IF(
            [Price Percentile Rank] >= 25,
            "25-50%",
            "Bottom 25% (Cheapest)"
        )
    )
)

// =====================================================================
// SECTION 4: VOLATILITY & STANDARD DEVIATION
// =====================================================================

// Price Volatility (Standard Deviation over last 6 cycles)
Price Volatility = 
STDEV.P(
    CALCULATE(
        [Current Average Price],
        FILTER(
            DATERANGE(
                MIN(dim_date[full_date]),
                MAX(dim_date[full_date])
            ),
            DATEDIFF(dim_date[full_date], MAX(dim_date[full_date]), DAY) <= 180
        )
    )
)

// Coefficient of Variation (Volatility as % of mean)
Price Volatility Coefficient = 
DIVIDE(
    [Price Volatility],
    [Current Average Price],
    0
) * 100

// Standard Deviation by Fuel Type
PMS Volatility = 
CALCULATE(
    [Price Volatility],
    FILTER(dim_fuel, dim_fuel[fuel_code] = "PMS")
)

Diesel Volatility = 
CALCULATE(
    [Price Volatility],
    FILTER(dim_fuel, dim_fuel[fuel_code] = "AGO")
)

Kerosene Volatility = 
CALCULATE(
    [Price Volatility],
    FILTER(dim_fuel, dim_fuel[fuel_code] = "IK")
)

// =====================================================================
// SECTION 5: REGIONAL & COMPARATIVE MEASURES
// =====================================================================

// National Average (used for comparisons)
National Average Price = 
CALCULATE(
    [Current Average Price],
    ALLSELECTED(dim_location[region])
)

// Regional Average
Regional Average Price = 
CALCULATE(
    [Current Average Price],
    ALLSELECTED(dim_location[town])
)

// Regional Premium/Discount vs National
Regional Premium % = 
DIVIDE(
    [Regional Average Price] - [National Average Price],
    [National Average Price],
    0
) * 100

// Min & Max prices in current selection
Min Price = 
MIN(fact_fuel_prices[price])

Max Price = 
MAX(fact_fuel_prices[price])

// Price Range
Price Range = 
[Max Price] - [Min Price]

// Quartile positions
Q1 Price = 
PERCENTILE.INC(
    fact_fuel_prices[price],
    0.25
)

Q2 Price (Median) = 
PERCENTILE.INC(
    fact_fuel_prices[price],
    0.5
)

Q3 Price = 
PERCENTILE.INC(
    fact_fuel_prices[price],
    0.75
)

// =====================================================================
// SECTION 6: AFFORDABILITY MEASURES
// =====================================================================

// Affordability Index (0-100, higher = more affordable)
Affordability Index = 
100 - 
DIVIDE(
    [Current Average Price] - [Min Price],
    [Max Price] - [Min Price],
    0
) * 100

// Alternative: Inverse of price percentile
Affordability Index (Inverse Percentile) = 
100 - [Price Percentile Rank]

// Affordability Score Label
Affordability Rating = 
IF(
    [Affordability Index] >= 70,
    "✓ Affordable",
    IF(
        [Affordability Index] >= 40,
        "⊙ Moderate",
        "✗ Unaffordable"
    )
)

// Fill-up Cost (50L tank)
Fill Up Cost (50L) = 
[Current Average Price] * 50

// Fill-up Cost MoM Change
Fill Up Cost Change (50L) = 
([Current Average Price] * 50) - ([Previous Average Price] * 50)

// =====================================================================
// SECTION 7: TREND & FORECASTING MEASURES
// =====================================================================

// Year-over-Year Change
YoY Change % = 
DIVIDE(
    [Current Average Price] - CALCULATE([Current Average Price], DATEADD(dim_date[full_date], -365, DAY)),
    CALCULATE([Current Average Price], DATEADD(dim_date[full_date], -365, DAY)),
    0
) * 100

// Trend Direction (3-cycle average comparison)
Trend Direction = 
VAR Current = [Current Average Price]
VAR Previous3CycleAvg = AVERAGE(
    CALCULATETABLE(
        [Current Average Price],
        FILTER(
            dim_date,
            DATEDIFF(dim_date[full_date], MAX(dim_date[full_date]), DAY) BETWEEN -90 AND -30
        )
    )
)
RETURN IF(Current > Previous3CycleAvg, "📈 Uptrend", IF(Current < Previous3CycleAvg, "📉 Downtrend", "➡️ Stable"))

// Simple Forecast (exponential smoothing)
Forecast Next Cycle = 
VAR LastPrice = [Current Average Price]
VAR PreviousPrice = [Previous Average Price]
VAR Smoothing = 0.5
RETURN LastPrice + (LastPrice - PreviousPrice) * Smoothing

// Forecast Range (±confidence interval)
Forecast Lower Bound (95% CI) = 
[Forecast Next Cycle] - (1.96 * [Price Volatility])

Forecast Upper Bound (95% CI) = 
[Forecast Next Cycle] + (1.96 * [Price Volatility])

// =====================================================================
// SECTION 8: ANOMALY DETECTION & OUTLIER MEASURES
// =====================================================================

// Z-Score (Standard Deviations from Mean)
Price Z Score = 
DIVIDE(
    [Current Average Price] - [National Average Price],
    [Price Volatility],
    0
)

// Flag if price is outlier (>2 std dev)
Is Outlier = 
IF(
    ABS([Price Z Score]) > 2,
    "⚠️ OUTLIER",
    "Normal"
)

// Deviation from Regional Mean
Deviation from Regional Mean % = 
DIVIDE(
    [Current Average Price] - [Regional Average Price],
    [Regional Average Price],
    0
) * 100

// High Variance Flag
Is High Variance Region = 
IF(
    [Regional Premium %] > 15,
    "🚩 High Premium",
    IF(
        [Regional Premium %] < -15,
        "🟢 Discount",
        "Normal"
    )
)

// Price Jump Detection (MoM > 5%)
Is Price Jump = 
IF(
    ABS([MoM Change Average %]) > 5,
    "🔴 PRICE JUMP",
    IF(
        ABS([MoM Change Average %]) > 2,
        "🟡 Significant Change",
        "🟢 Stable"
    )
)

// =====================================================================
// SECTION 9: DATA QUALITY & COMPLETENESS MEASURES
// =====================================================================

// Count of towns with data in current cycle
Towns Reported = 
DISTINCTCOUNT(dim_location[location_key])

// Data Completeness Percentage (223 = total expected towns)
Data Completeness % = 
DIVIDE(
    [Towns Reported],
    223,
    0
) * 100

// Data Quality Score (0-100)
Data Quality Score = 
[Data Completeness %] * 0.7 + 
(100 - [Price Volatility Coefficient]) * 0.3

// Count of records (fact rows)
Record Count = 
COUNTA(fact_fuel_prices[price])

// Count of missing values (nulls)
Null Count = 
CALCULATE(
    COUNTA(fact_fuel_prices[price]),
    FILTER(fact_fuel_prices, fact_fuel_prices[price] = BLANK())
)

// Data Freshness (days since last update)
Data Freshness Days = 
INT(NOW() - MAX(dim_date[full_date]))

// =====================================================================
// SECTION 10: REGIONAL AGGREGATION MEASURES
// =====================================================================

// Count of regions
Region Count = 
DISTINCTCOUNT(dim_location[region])

// Count of locations per region
Locations Per Region = 
DISTINCTCOUNT(dim_location[town])

// Average price by region
Region Average Price = 
AVERAGE(fact_fuel_prices[price])

// Regional price variance
Region Price Variance = 
VAR RegionalAvg = [Region Average Price]
RETURN STDEV.P(
    CALCULATE(
        fact_fuel_prices[price],
        FILTER(dim_location, dim_location[region] = SELECTEDVALUE(dim_location[region]))
    )
)

// Most expensive region (by average)
Most Expensive Region = 
CALCULATE(
    [Current Average Price],
    TOPN(1, VALUES(dim_location[region]), [Current Average Price], DESC)
)

// Cheapest region (by average)
Cheapest Region = 
CALCULATE(
    [Current Average Price],
    TOPN(1, VALUES(dim_location[region]), [Current Average Price], ASC)
)

// Regional Price Difference (spread)
Regional Price Spread = 
[Most Expensive Region] - [Cheapest Region]

// =====================================================================
// SECTION 11: FUEL TYPE AGGREGATION MEASURES
// =====================================================================

// Count of fuel types
Fuel Type Count = 
DISTINCTCOUNT(dim_fuel[fuel_key])

// Fuel-specific prices (alternative calc)
Fuel Average Price = 
AVERAGE(fact_fuel_prices[price])

// Price difference between fuel types
PMS vs Diesel Diff = 
[Current PMS Price] - [Current Diesel Price]

PMS vs Kerosene Diff = 
[Current PMS Price] - [Current Kerosene Price]

Diesel vs Kerosene Diff = 
[Current Diesel Price] - [Current Kerosene Price]

// Most expensive fuel type
Most Expensive Fuel = 
MAXX(
    SUMMARIZE(dim_fuel, dim_fuel[fuel_code]),
    CALCULATE([Current Average Price])
)

// =====================================================================
// SECTION 12: TIME SERIES & HISTORICAL MEASURES
// =====================================================================

// 3-Cycle Average
3 Cycle Average = 
AVERAGE(
    CALCULATETABLE(
        [Current Average Price],
        FILTER(
            dim_date,
            DATEDIFF(dim_date[full_date], MAX(dim_date[full_date]), DAY) <= 90
        )
    )
)

// 6-Cycle Average
6 Cycle Average = 
AVERAGE(
    CALCULATETABLE(
        [Current Average Price],
        FILTER(
            dim_date,
            DATEDIFF(dim_date[full_date], MAX(dim_date[full_date]), DAY) <= 180
        )
    )
)

// 12-Cycle Average (Yearly)
12 Cycle Average = 
AVERAGE(
    CALCULATETABLE(
        [Current Average Price],
        FILTER(
            dim_date,
            DATEDIFF(dim_date[full_date], MAX(dim_date[full_date]), DAY) <= 365
        )
    )
)

// =====================================================================
// SECTION 13: PRICE INDEX MEASURES
// =====================================================================

// Base Index (100 = first date in filter)
Price Index (Base = First Date) = 
DIVIDE(
    [Current Average Price],
    CALCULATE(
        [Current Average Price],
        FILTER(
            ALL(dim_date),
            dim_date[date_key] = MIN(dim_date[date_key])
        )
    ),
    1
) * 100

// Index vs 12-month average
Price Index (Base = 12M Avg) = 
DIVIDE(
    [Current Average Price],
    [12 Cycle Average],
    1
) * 100

// =====================================================================
// SECTION 14: SUMMARY & NARRATIVE MEASURES
// =====================================================================

// Key Insight: Highest location
Highest Price Location = 
CALCULATE(
    CONCATENATEX(
        TOPN(1, VALUES(dim_location[town]), [Current Average Price], DESC),
        dim_location[town],
        ", "
    )
)

Highest Price Amount = 
CALCULATE(
    [Current Average Price],
    TOPN(1, VALUES(dim_location[town]), [Current Average Price], DESC)
)

// Key Insight: Lowest location
Lowest Price Location = 
CALCULATE(
    CONCATENATEX(
        TOPN(1, VALUES(dim_location[town]), [Current Average Price], ASC),
        dim_location[town],
        ", "
    )
)

Lowest Price Amount = 
CALCULATE(
    [Current Average Price],
    TOPN(1, VALUES(dim_location[town]), [Current Average Price], ASC)
)

// Auto-generated narrative
Key Insight Narrative = 
"National avg: " & FORMAT([Current Average Price], "0.0") & " KES | " &
"MoM: " & FORMAT([MoM Change Average %], "+0.0%; -0.0%; 0.0%") & " | " &
"Range: " & FORMAT([Price Range], "0.0") & " KES | " &
"Data: " & FORMAT([Data Completeness %], "0%")

// =====================================================================
// SECTION 15: SUPPLY CHAIN & COST ANALYSIS
// =====================================================================

// Estimated transport cost (assuming Nairobi is base)
Estimated Transport Premium = 
IF(
    SELECTEDVALUE(dim_location[region]) = "Nairobi",
    0,
    [Regional Premium %] * 0.6  // Assume 60% of regional premium is transport
)

// Price deviation likely due to transport
Transport Cost Impact = 
[Current Average Price] - CALCULATE(
    [Current Average Price],
    FILTER(dim_location, dim_location[region] = "Nairobi")
)

// =====================================================================
// SECTION 16: PERFORMANCE INDICATORS
// =====================================================================

// Color indicator for KPI (for conditional formatting)
Price Status Color = 
IF(
    [MoM Change Average %] > 5,
    "Red",
    IF(
        [MoM Change Average %] > 2,
        "Orange",
        IF(
            [MoM Change Average %] < -2,
            "Green",
            "Gray"
        )
    )
)

// Dashboard Status
Dashboard Status = 
IF(
    [Data Completeness %] >= 95,
    "🟢 Healthy",
    IF(
        [Data Completeness %] >= 80,
        "🟡 Warning",
        "🔴 Error"
    )
)
